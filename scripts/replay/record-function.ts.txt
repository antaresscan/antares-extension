// THROWAWAY recorder: copy to api/replay-record.ts in a PREVIEW deployment, never commit it under api/ and never deploy it to
// production. It runs the REAL /api/scan handler for one mint and returns every outgoing HTTP exchange of that scan
// (gzipped, base64), so the scan can be replayed offline (see scripts/replay/README.md).
//
// Each recording starts from a clean engine state: the instance is warm and shared by every token captured, and the engine's
// circuit breakers (5 failures in 30 s block a host for 30 s) or the Solscan pause (10 min after a 401) would otherwise leak
// from one token into the next, so that later tokens never even ask the host that failed before. A request that FAILS
// (timeout, abort) is recorded too, as a failure, so the replay fails it in the same way.
//
// Secrets never leave the function: credentials in URL query strings are redacted, headers are not recorded, the GoPlus token
// exchange (its body carries the app key and its signature) and the Gemini calls (key in the URL, the prompt in the body) are
// not recorded at all. Deploy it with Redis and Sentry blanked, so nothing is written to production's database.
//
// The insider graph calls (Helius getSignaturesForAddress and enhanced transactions) run for real but are not recorded: they only
// feed the Critical Actors cards and the holder activity of the response, never the verdict, and they would weigh more than all the
// other calls together (see __tests__/replay/harness.ts, insiderGraphAnswer).
import type { VercelRequest, VercelResponse } from "@vercel/node";
import { gzipSync } from "node:zlib";
import scan from "./scan";
import { _resetCircuitBreakersForTests } from "./_lib/http";
import { _resetSolscanBackoffForTests } from "./_lib/fetchers";
import { _resetGoPlusAuthForTests } from "./_lib/goplus-auth";
import { _resetHeliusAuthForTests, heliusRpc } from "./_lib/helius";
import { HOLDER_MAX_PAGES, HOLDER_PAGE_LIMIT } from "./_lib/constants";

interface Exchange { method: string; url: string; body: string | null; status: number; contentType: string | null; text: string; error?: string; auth?: boolean }

/**
 * Helius answers the deep pages of a big token's holder list (getTokenAccounts, 10 pages asked in parallel with a 4 s timeout) too
 * slowly the FIRST time: in a capture, about a third of the scans of tokens with thousands of holders lost pages 3 to 10, and the
 * second try, a few seconds later, got them all. So the pages are asked once beforehand, with a long timeout and without being
 * recorded, and the recorded scan then sees what a scan sees when Helius is warm: the complete data the corpus is meant to hold.
 */
async function warmHolderPages(mint: string): Promise<void> {
  const key = process.env.HELIUS_API_KEY;
  if (!key) return;
  const page = (p: number) => heliusRpc<{ result?: { token_accounts?: unknown[] } }>(key, { jsonrpc: "2.0", id: "warm", method: "getTokenAccounts", params: { mint, limit: HOLDER_PAGE_LIMIT, page: p } }, 25_000, 0);
  const first = await page(1);
  if ((first?.result?.token_accounts?.length ?? 0) < HOLDER_PAGE_LIMIT) return;
  await Promise.all(Array.from({ length: HOLDER_MAX_PAGES - 1 }, (_, i) => page(i + 2)));
}

const redact = (url: string): string => url.replace(/([?&](?:api-key|apikey|key|token|access_token)=)[^&#]*/gi, "$1REDACTED");
const NOT_RECORDED = [/generativelanguage\.googleapis\.com/i, /gopluslabs\.io\/api\/v1\/token(\?|$)/i, /upstash\.io/i, /sentry\.io/i, /api\.helius\.xyz\/v0\/transactions/i];
const isInsiderGraphCall = (href: string, body: string | null): boolean =>
  /helius-rpc\.com/i.test(href) && body !== null && /"method"\s*:\s*"getSignaturesForAddress"/.test(body);

export default async function handler(req: VercelRequest, res: VercelResponse): Promise<void> {
  const ca = String(req.query.ca ?? "");
  _resetCircuitBreakersForTests();
  _resetSolscanBackoffForTests();
  _resetGoPlusAuthForTests();
  _resetHeliusAuthForTests();

  if (req.query.warm !== "0") {
    await warmHolderPages(ca).catch(() => undefined);
    _resetCircuitBreakersForTests();
    _resetHeliusAuthForTests();
  }

  const exchanges: Exchange[] = [];
  const realFetch = globalThis.fetch;
  globalThis.fetch = (async (input: RequestInfo | URL, init?: RequestInit) => {
    const href = typeof input === "string" ? input : input instanceof URL ? input.href : (input as Request).url;
    const body = init?.body == null ? null : String(init.body);
    const recorded = !NOT_RECORDED.some((re) => re.test(href)) && !isInsiderGraphCall(href, body);
    const auth = new Headers(init?.headers ?? (typeof input === "object" && "headers" in input ? (input as Request).headers : undefined)).has("authorization");
    const base = { method: (init?.method ?? "GET").toUpperCase(), url: redact(href), body, ...(auth ? { auth: true } : {}) };
    let response: Response;
    try {
      response = await realFetch(input, init);
    } catch (e) {
      if (recorded) exchanges.push({ ...base, status: 0, contentType: null, text: "", error: (e as Error).name || "Error" });
      throw e;
    }
    if (recorded) {
      const text = await response.clone().text();
      exchanges.push({ ...base, status: response.status, contentType: response.headers.get("content-type"), text });
    }
    return response;
  }) as typeof fetch;

  const startedAt = Date.now();
  let status = 200;
  let result: unknown = null;
  const fakeRes = {
    setHeader: () => fakeRes,
    status: (code: number) => { status = code; return fakeRes; },
    json: (data: unknown) => { result = data; return fakeRes; },
    end: () => fakeRes,
  };
  try {
    await scan(
      { method: "GET", query: { ca, fresh: "1" }, headers: { origin: "https://dexscreener.com", "x-forwarded-for": "203.0.113.7" }, socket: { remoteAddress: "203.0.113.7" } } as unknown as VercelRequest,
      fakeRes as unknown as VercelResponse,
    );
  } finally {
    globalThis.fetch = realFetch;
  }
  res.setHeader("Cache-Control", "no-store");
  res.json({ ca, startedAt, finishedAt: Date.now(), status, result, exchanges: gzipSync(Buffer.from(JSON.stringify(exchanges))).toString("base64"), exchangeCount: exchanges.length });
}
